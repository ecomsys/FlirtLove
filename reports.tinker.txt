Создать 10 случайных жалоб


use App\Models\Report;
use App\Models\User;
use App\Models\Photo;

$users = User::all();
$reporter = $users->first();

$reasons = [
    'Спамит в чате',
    'Оскорбительное поведение',
    'Фейковый профиль',
    'Нарушает правила',
    'Неприемлемый контент',
    'Мошенничество',
    'Агрессивное поведение',
    'Спам в личных сообщениях',
];

$statuses = ['pending', 'pending', 'pending', 'resolved', 'rejected'];

for ($i = 0; $i < 10; $i++) {
    $reported = $users->skip(rand(1, $users->count() - 1))->first();
    $type = rand(0, 1) ? 'user' : 'photo';
    $photoId = null;
    
    if ($type === 'photo') {
        $photo = Photo::inRandomOrder()->first();
        if ($photo) {
            $photoId = $photo->id;
        } else {
            $type = 'user';
        }
    }
    
    Report::create([
        'user_id' => $reporter->id,
        'reported_user_id' => $reported->id,
        'photo_id' => $photoId,
        'reason' => $reasons[array_rand($reasons)] . ' #' . ($i + 1),
        'status' => $statuses[array_rand($statuses)],
        'type' => $type,
        'created_at' => now()->subDays(rand(0, 10)),
    ]);
}

Report::count(); // 10

use App\Models\Report;
use App\Models\User;
use App\Models\Photo;

$users = User::all();
$reporter = $users->first();

if ($users->count() < 2) {
    echo "❌ Нужно минимум 2 пользователя!\n";
    return;
}

$reasonsUser = [
    'Оскорбляет других пользователей в чате',
    'Профиль выглядит фейковым',
    'Спамит ссылками на посторонние сайты',
    'Выдает себя за другого человека',
    'Рассылает неприемлемый контент',
    'Нарушает правила сообщества',
    'Пытается мошенничать, просит деньги',
    'Создает несколько фейковых аккаунтов',
];

$reasonsPhoto = [
    'На фото явно не настоящий человек',
    'Фото взято из интернета',
    'На фото неприемлемый контент',
    'Фото слишком откровенное',
    'Фото не соответствует описанию профиля',
];

$statuses = ['pending', 'pending', 'pending', 'resolved', 'rejected'];

for ($i = 0; $i < 30; $i++) {
    $reported = $users->skip(rand(1, $users->count() - 1))->first();
    $type = rand(0, 1) ? 'user' : 'photo';
    $photoId = null;
    
    if ($type === 'photo') {
        $photo = Photo::inRandomOrder()->first();
        if ($photo) {
            $photoId = $photo->id;
            $reason = $reasonsPhoto[array_rand($reasonsPhoto)];
        } else {
            $type = 'user';
            $reason = $reasonsUser[array_rand($reasonsUser)];
        }
    } else {
        $reason = $reasonsUser[array_rand($reasonsUser)];
    }
    
    $status = $statuses[array_rand($statuses)];
    
    Report::create([
        'user_id' => $reporter->id,
        'reported_user_id' => $reported->id,
        'photo_id' => $photoId,
        'reason' => $reason . ' (жалоба #' . ($i + 1) . ')',
        'status' => $status,
        'type' => $type,
        'created_at' => now()->subDays(rand(0, 14))->subHours(rand(0, 24)),
    ]);
}

$pending = Report::where('status', 'pending')->count();
$resolved = Report::where('status', 'resolved')->count();
$rejected = Report::where('status', 'rejected')->count();

echo "✅ Создано " . Report::count() . " жалоб:\n";
echo "   ⏳ Ожидают: $pending\n";
echo "   ✅ Решены: $resolved\n";
echo "   ❌ Отклонены: $rejected\n";
